§ Quickstart

Live in 30 seconds.

from openai import OpenAI

client = OpenAI(
    base_url="https://api.vigas.ai/v1",
    api_key="sk-vg-v1-…",
)
resp = client.chat.completions.create(
    model="anthropic/claude-opus-5.5",  # or "vigas/auto"
    messages=[{"role": "user", "content": "Hello"}],
)
 
import OpenAI from "openai";

const client = new OpenAI({
  baseURL: "https://api.vigas.ai/v1",
  apiKey: process.env.VIGAS_API_KEY,
});
const resp = await client.chat.completions.create({
  model: "vigas/auto",
  messages: [{ role: "user", content: "Hello" }],
});
 
curl https://api.vigas.ai/v1/chat/completions \

  -H "Content-Type: application/json" \
  -d '{"model": "vigas/auto", "messages": [{"role": "user", "content": "Hello"}]}'
